### 8.3 EventBus test 示例

```typescript
import { describe, it, expect, beforeEach } from 'vitest'
import { EventBus } from '../../electron/runtime/bridge/EventBus'

describe('EventBus', () => {
  let bus: EventBus
  beforeEach(() => {
    bus = EventBus.getInstance()
    bus.clear()
  })

  it('subscribe + publish 通知订阅者', async () => {
    let received: unknown = null
    bus.subscribe('test:topic', (payload) => { received = payload })
    await bus.publish('test:topic', { msg: 'hello' }, 'test-source')
    expect(received).toEqual({ msg: 'hello' })
  })

  it('unsubscribe 取消订阅', async () => {
    let count = 0
    const unsub = bus.subscribe('test:topic', () => { count += 1 })
    await bus.publish('test:topic', 'a')
    unsub()
    await bus.publish('test:topic', 'b')
    expect(count).toBe(1)
  })

  it('historyOf 记录最近事件', async () => {
    await bus.publish('test:topic', 'a')
    await bus.publish('test:topic', 'b')
    const history = bus.historyOf('test:topic')
    expect(history.length).toBe(2)
    expect(history[0].payload).toBe('a')
    expect(history[1].payload).toBe('b')
  })

  it('clear 清空订阅与历史', async () => {
    bus.subscribe('test:topic', () => {})
    await bus.publish('test:topic', 'a')
    bus.clear()
    expect(bus.historyOf().length).toBe(0)
  })
})
```
